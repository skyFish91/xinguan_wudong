# 乌东文旅平台 · 公共模块（公共层）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有 Cool Admin 8.0 脚手架上实现「乌东文旅」平台的公共层能力（统一订单中心、购物车、支付、收藏、消息、敏感词、搜索、商家体系、公共常量与工具），为后续衣食住行社区等业务板块冻结可复用的契约。

**Architecture:** 沿用 Cool Admin 8.0（Midway v3.20.3 + Koa + TypeORM `synchronize:true`）的模块化单体结构。每个公共能力是一个 `src/modules/{name}` 模块（`config.ts` + `entity/` + `controller/{admin,app}/` + `service/`）。纯逻辑（金额分、订单号、订单状态机）下沉到 `src/common/` 用 Jest 做 TDD。业务板块（m1–m5）通过注入公共层 Service（`OrderService`/`CartService`/`PayService`/`FavoriteService`/`MessageService`/`SensitiveService`）复用能力，禁止相互 import。

**Tech Stack:** Midway.js v3.20.3（Koa 模式）· TypeScript ~5.8 · TypeORM（`@midwayjs/typeorm`）· MySQL 8.0（库名 `wudong`）· Jest + ts-jest（纯函数单测）。

**Spec:** `D:/workspace/ClaudeCode/xinguan_wudong/乌东文旅平台-设计文档（最终整合版）.md`（以下简称「设计文档」）。本计划实现其第二部分（§6–§13 公共能力）与第三/四部分的公共表与公共接口；业务板块 §22–§27 由后续 subagent 计划实现。

## 全局约束（Global Constraints）

以下每一条对所有任务隐含生效，执行者不得违反：

- 金额一律存「分」、`INT`、`unsigned:true`；接口层进出用 `src/common/money.ts` 的 `yuanToFen`/`fenToYuan` 转换，禁止直接拼小数。
- 所有新实体 `extends BaseEntity`（`src/modules/base/entity/base.ts`，含 `id`/`createTime`/`updateTime`/`tenantId`）。
- 所有新 Controller 用 `@CoolController({...})` 或 `@Get/@Post('...', { summary })`，方法返回 `this.ok(data)`；错误统一 `throw new BizException(code, msg)`，**禁止**手写 `{code,message,data}`。
- 业务错误码按设计文档 §6.2 号段：公共 10000–19999，用户 20000–29999，衣 30000–39999，食 40000–49999，住 50000–59999，行 60000–69999，社区 70000–79999，管理 80000–89999，订单/支付/库存 90000–99999。
- 跨板块红线：业务板块只允许 import `src/common/*` 与公共层 Service（`src/modules/{order,cart,pay,favorite,message,sensitive,merchant}/...`），**禁止** import 兄弟业务板块的 Service（设计文档 §34.3）。
- 库存/房态/餐位/门票一律「条件更新 + 乐观锁」，禁止「先查再改」（设计文档 §13）。
- 订单状态只能走 `OrderService.changeStatus()`，主表 `status` 写权限只归 OrderService（设计文档 §8.3）。
- 每个模块需 `config.ts`（`ModuleConfig`），`name`/`description` 用中文。
- 新增枚举统一放 `src/common/constants/`；新增实体/服务遵循 Cool Admin 目录约定。

```typescript
/**
 * 金额工具：全平台金额统一用「分」(整数)存储与传输。
 * 设计文档 §14：禁止 DECIMAL/FLOAT；§8.1：金额字段存分。
 */
export function yuanToFen(yuan: number | string): number {
  const n = typeof yuan === 'string' ? parseFloat(yuan) : yuan;
  if (Number.isNaN(n)) {
    throw new Error('金额格式错误');
  }
  return Math.round(n * 100);
}

export function fenToYuan(fen: number): number {
  return Math.round(fen) / 100;
}
```

```typescript
/**
 * 订单号生成：yyyyMMddHHmmss + 6 位随机。
 * 幂等以 clientRequestId 为准（Task 4），order_no 仅做展示与追溯。
 */
export function genOrderNo(prefix = 'WD'): string {
  const now = new Date();
  const p = (n: number, w = 2) => String(n).padStart(w, '0');
  const ts =
    `${now.getFullYear()}${p(now.getMonth() + 1)}${p(now.getDate())}` +
    `${p(now.getHours())}${p(now.getMinutes())}${p(now.getSeconds())}`;
  const rand = Math.floor(Math.random() * 1000000)
    .toString()
    .padStart(6, '0');
  return `${prefix}${ts}${rand}`;
}
```

```typescript
import { ErrorCode } from './constants/error-code';

/**
 * 业务异常：携带号段内的业务错误码。
 * Cool Admin 的 CoolExceptionFilter 会把 err.status 映射到响应 { code, message }，

export class BizException extends Error {
  status: number;
  statusCode?: number;

  constructor(code: number, message: string, statusCode?: number) {
    super(message);
    this.name = 'BizException';
    this.status = code;
    this.statusCode = statusCode;
  }
}

export function bizError(code: ErrorCode, message: string): BizException {
  return new BizException(code, message);
}
```

```typescript
 * 状态变更唯一入口是 OrderService.changeStatus()（Task 4）。
 */
export enum OrderStatus {
  PENDING = 'PENDING',
  PAID = 'PAID',
  CONFIRMED = 'CONFIRMED',
  ONGOING = 'ONGOING',
  FINISHED = 'FINISHED',
  REVIEWED = 'REVIEWED',
  CANCELLED = 'CANCELLED',
  CLOSED = 'CLOSED',
  REFUNDED = 'REFUNDED',
}

export const OrderTransition: Record<string, OrderStatus[]> = {
  [OrderStatus.PENDING]: [OrderStatus.PAID, OrderStatus.CANCELLED, OrderStatus.CLOSED],
  [OrderStatus.PAID]: [OrderStatus.CONFIRMED, OrderStatus.REFUNDED],
  [OrderStatus.CONFIRMED]: [OrderStatus.ONGOING, OrderStatus.REFUNDED],
  [OrderStatus.ONGOING]: [OrderStatus.FINISHED],
  [OrderStatus.FINISHED]: [OrderStatus.REVIEWED, OrderStatus.REFUNDED],
};

export function canTransit(from: OrderStatus, to: OrderStatus): boolean {
  return OrderTransition[from]?.includes(to) ?? false;
}
```

```typescript
 */
@Provide()
export class MerchantScopeService {
  @InjectEntityModel(MerchantEntity)
  merchantEntity: Repository<MerchantEntity>;

  @Inject()
  ctx;

  /**
   * 若当前后台用户是商家角色，返回其 merchantId；否则返回 undefined（平台管理员看全量）。
   */
  async getMerchantId(ctx?: any): Promise<number | undefined> {
    const c = ctx || this.ctx;
    const admin = c?.admin; // Cool Admin 后台鉴权注入 ctx.admin
    if (!admin || admin.roleIds?.includes(1)) {
      return undefined; // 超级管理员/平台角色看全量
    }
    const merchant = await this.merchantEntity.findOneBy({
      ownerUserId: admin.userId,
      status: 1,
    });
    return merchant?.id;
  }
}
```

```typescript
  @InjectEntityModel(OrderExtSeatEntity)

  /**

   * @param orderId 订单ID
   * @param bizType 业务类型
   * @param ext 板块传入的扩展字段
   * @param em 事务 EntityManager（在 OrderService 事务内调用）
   */
  async write(orderId: number, bizType: string, ext: Record<string, any>, em: any): Promise<void> {
    const payload = { orderId, ...ext };
    switch (bizType) {
      case BizType.GOODS:
        await em.save(OrderExtGoodsEntity, payload);
        break;
      case BizType.SEAT:
        await em.save(OrderExtSeatEntity, payload);
        break;
      case BizType.STAY:
        await em.save(OrderExtStayEntity, payload);
        break;
      case BizType.TICKET:
      case BizType.ROUTE:
        await em.save(OrderExtTicketEntity, payload);
        break;
      default:
        throw new Error(`未知 bizType: ${bizType}`);
    }
  }
```

```typescript
    const exist = await this.orderEntity.findOneBy({ clientRequestId: params.clientRequestId });
    if (exist) {
      return { orderId: exist.id, orderNo: exist.orderNo, payAmount: exist.payAmount, status: exist.status };
    }

    const orderNo = genOrderNo('WD');
    // 2. 事务：主表 + 明细 + 扩展表 + 日志（用 repository.manager 获取事务管理器）
    return this.orderEntity.manager.transaction(async (em: any) => {
        const order = await em.save(OrderEntity, {
          orderNo,
          userId: params.userId,
          merchantId: params.merchantId,
          bizType: params.bizType,
          totalAmount: params.totalAmount,
          payAmount: params.totalAmount,
          status: OrderStatus.PENDING,
          payStatus: 0,
          clientRequestId: params.clientRequestId,
          addressSnapshot: params.addressSnapshot,
          remark: params.remark,
        });
        for (const it of params.items || []) {
          await em.save(OrderItemEntity, { orderId: order.id, ...it });
        }
        await this.orderExtService.write(order.id, params.bizType, params.ext || {}, em);
        await em.save(OrderLogEntity, {
          orderId: order.id,
          orderNo,
          fromStatus: null,
          toStatus: OrderStatus.PENDING,
          remark: '创建订单',
        });
        return { orderId: order.id, orderNo, payAmount: order.payAmount, status: order.status };
      });
  }

  /**
   * 状态变更唯一入口（设计文档 §8.3）：校验流转 + 乐观更新 + 日志 + 事件
   */
  async changeStatus(orderId: number, to: string, opts?: { reason?: string; operatorId?: number }) {
    const order = await this.orderEntity.findOneBy({ id: orderId });
    if (!order) throw bizError(ErrorCode.ORDER_NOT_FOUND, '订单不存在');
    if (!canTransit(order.status as OrderStatus, to as OrderStatus)) {
      throw bizError(ErrorCode.ORDER_STATUS_INVALID, `订单状态不允许 ${order.status} -> ${to}`);
    }
    // 乐观更新防并发重复变更
    const r = await this.orderEntity.update(
      { id: orderId, status: order.status },
      { status: to }
    );
    if (r.affected === 0) throw bizError(ErrorCode.ORDER_STATUS_INVALID, '订单状态已变更，请刷新');

    await this.logEntity.save({
      orderId,
      orderNo: order.orderNo,
      fromStatus: order.status,
      toStatus: to,
      operatorId: opts?.operatorId,
      remark: opts?.reason,
    });
    // 发布事件，供各板块订阅（扣房态/生成电子票等）
    this.eventManager.emit(
      `order.${to.toLowerCase()}`,
      { orderNo: order.orderNo, bizType: order.bizType, orderId }
    );
  }

  /**
   * 取消订单
   */
  async cancel(orderId: number, userId: number) {
    const order = await this.orderEntity.findOneBy({ id: orderId });
    if (!order || order.userId !== userId) throw bizError(ErrorCode.ORDER_NOT_FOUND, '订单不存在');
    await this.changeStatus(orderId, OrderStatus.CANCELLED, { reason: '用户取消', operatorId: userId });
  }

  /**
   * 申请退款（penaltyRate 0~1，由各板块按取消政策计算后传入，设计文档 §8.4）
   */
  async refund(orderId: number, userId: number, penaltyRate: number, reason?: string) {
    const order = await this.orderEntity.findOneBy({ id: orderId });
    if (!order || order.userId !== userId) throw bizError(ErrorCode.ORDER_NOT_FOUND, '订单不存在');
    if (![OrderStatus.PAID, OrderStatus.CONFIRMED, OrderStatus.FINISHED].includes(order.status as OrderStatus)) {
      throw bizError(ErrorCode.ORDER_STATUS_INVALID, '当前状态不可退款');
    }
    const amount = Math.round(order.payAmount * (1 - penaltyRate));
    await this.refundEntity.save({
      orderId,
      refundNo: genOrderNo('RF'),
      amount,
      penaltyRate,
      reason,
      status: 0,
    });
    await this.changeStatus(orderId, OrderStatus.REFUNDED, { reason: reason || '退款', operatorId: userId });
    return { refundNo: this.refundEntity.id, amount };
  }

  /**
   * 用户订单分页
   */
  async list(userId: number, query: any) {
    const qb = this.orderEntity
      .createQueryBuilder('o')
      .where('o.userId = :userId', { userId });
    if (query.bizType) qb.andWhere('o.bizType = :bizType', { bizType: query.bizType });
    if (query.status) qb.andWhere('o.status = :status', { status: query.status });
    qb.orderBy('o.id', 'DESC');
    return this.entityRenderPage(qb, query);
  }

  /**
   * 订单详情（主表 + 明细 + 扩展表）
   */
  async detail(orderId: number, userId?: number) {
    const where: any = { id: orderId };
    if (userId) where.userId = userId;
    const order = await this.orderEntity.findOneBy(where);
    if (!order) throw bizError(ErrorCode.ORDER_NOT_FOUND, '订单不存在');
    const items = await this.orderItemEntity.findBy({ orderId });
    const ext = await this.loadExt(orderId, order.bizType);
    return { ...order, items, ext };
  }

  private async loadExt(orderId: number, bizType: string) {
    const map: any = {
      GOODS: 'extGoods',
      SEAT: 'extSeat',
      STAY: 'extStay',
      TICKET: 'extTicket',
      ROUTE: 'extTicket',
    };
    const repo = (this.orderExtService as any)[map[bizType]];
    return repo ? repo.findOneBy({ orderId }) : null;
  }

  /**
   * 从购物车结算（按 merchantId 拆单，Task 5 提供）
   * 此方法在 Task 5 完成后实现，先留契约。
   */
  async createFromCart(userId: number, cartItemIds: number[], addressId: number, clientRequestId: string) {
    throw new Error('createFromCart 在购物车任务中实现');
  }
}
```

```typescript
import { Inject, Provide } from '@midwayjs/core';
import { BaseService } from '@cool-midway/core';
import { InjectEntityModel } from '@midwayjs/typeorm';
import { Repository } from 'typeorm';
import { CartEntity } from '../entity/cart';
import { CartItemEntity } from '../entity/cartItem';
import { OrderService } from '../../order/service/order';
import { CartItemType } from '../../../common/constants/biz-type';
import { bizError } from '../../../common/biz-error';
import { ErrorCode } from '../../../common/constants/error-code';

/**
 * 统一购物车服务（设计文档 §9）
 * 加购库存/规格校验由各板块自己的 Service 做，本服务只负责存取。
 */
@Provide()
export class CartService extends BaseService {
  @InjectEntityModel(CartEntity) cartEntity: Repository<CartEntity>;
  @InjectEntityModel(CartItemEntity) cartItemEntity: Repository<CartItemEntity>;

  @Inject() orderService: OrderService;

  async getCart(userId: number) {
    let cart = await this.cartEntity.findOneBy({ userId });
    if (!cart) {
      cart = await this.cartEntity.save({ userId });
    }
    return cart;
  }

  async add(userId: number, item: { itemType: string; targetId: number; skuId?: number; quantity: number }) {
    if (![CartItemType.PRODUCT, CartItemType.FARM].includes(item.itemType as CartItemType)) {
      throw bizError(ErrorCode.CART_ITEM_INVALID, '购物车仅支持商品/特产');
    }
    const cart = await this.getCart(userId);
    const exist = await this.cartItemEntity.findOneBy({
      cartId: cart.id,
      itemType: item.itemType,
      targetId: item.targetId,
      skuId: item.skuId ?? null,
    });
    if (exist) {
      await this.cartItemEntity.update(exist.id, { quantity: exist.quantity + item.quantity });
    } else {
      await this.cartItemEntity.save({ cartId: cart.id, ...item });
    }
  }

  async update(userId: number, itemId: number, quantity: number) {
    const cart = await this.getCart(userId);
    await this.cartItemEntity.update({ id: itemId, cartId: cart.id }, { quantity });
  }

  async remove(userId: number, itemIds: number[]) {
    const cart = await this.getCart(userId);
    await this.cartItemEntity.delete({ cartId: cart.id, id: In(itemIds) });
  }

  async list(userId: number) {
    const cart = await this.getCart(userId);
    const items = await this.cartItemEntity.findBy({ cartId: cart.id });
    return items;
  }

  /**
   * 从购物车结算（设计文档 §9）：按 merchantId 拆单。
   * 各条目需附带 merchantId；这里先按 targetId 所属商家在业务板块解析。
   * 简化实现：由调用方传入 item->merchantId 映射（后续业务板块扩展）。
   */
  async createFromCart(userId: number, cartItemIds: number[], addressId: number, clientRequestId: string) {
    throw new Error('需要业务板块提供商品/特产 -> merchantId 映射后实现');
  }
}
```

> `createFromCart` 拆单需要「商品/特产 → 商家」映射，这一映射只有衣/食板块知道。为避免公共层反向依赖业务板块，实现策略（本计划内先冻结契约、留桩）：由业务板块在调用前完成「校验 + 组装 `CreateOrderParams[]`」，公共层 `OrderService.createFromCart` 仅接收已组装好的参数数组。因此把 `createFromCart` 改为**按已分组参数数组调用 `OrderService.create`** 的循环即可，见 Step 4。

```typescript
   * 从购物车结算（设计文档 §9）：接收已分组的订单参数，逐单创建。
   * 分组（按 merchantId 拆单）由业务板块完成，公共层只负责事务化创建。

      results.push(r);
    }
    return results;
  }
```

```typescript
import { Inject, Provide } from '@midwayjs/core';
import { BaseService } from '@cool-midway/core';
import { MockPayProvider } from '../provider/mockPayProvider';
import { OrderService } from '../../order/service/order';
import { OrderStatus } from '../../../common/order-status';

/**
 * 统一支付服务（设计文档 §10）
 */
@Provide()
export class PayService extends BaseService {
  @Inject() provider: MockPayProvider; // 后续按配置切换 WechatPayProvider
  @Inject() orderService: OrderService;

  /**
   * 发起支付（mock 直接改状态为 PAID）
   */
  async pay(orderNo: string) {
    const order = await this.orderService.orderEntity.findOneBy({ orderNo });
    if (!order) throw new Error('订单不存在');
    if (order.status !== OrderStatus.PENDING) {
      return { paid: false, reason: '订单已处理' };
    }
    const res = await this.provider.createPayment(orderNo, order.payAmount);
    await this.orderService.changeStatus(order.id, OrderStatus.PAID, { reason: '支付成功(mock)' });
    return { paid: true, payNo: res.payNo };
  }

  /**
   * 支付回调幂等（设计文档 §29.3）：重复回调不重复改状态
   */
  async handleNotify(payload: any) {
    const order = await this.orderService.orderEntity.findOneBy({ orderNo: payload.out_trade_no });
    if (!order) return { code: 'FAIL' };
    if (order.status !== OrderStatus.PENDING) {
      return { code: 'SUCCESS' }; // 幂等：已处理
    }
    await this.orderService.changeStatus(order.id, OrderStatus.PAID, { reason: '支付回调' });
    return { code: 'SUCCESS' };
  }
}
```

```typescript
 */
@Provide()
export class FavoriteService extends BaseService {
  @InjectEntityModel(FavoriteEntity) favoriteEntity: Repository<FavoriteEntity>;

  async toggle(userId: number, targetType: string, targetId: number) {
    const exist = await this.favoriteEntity.findOneBy({ userId, targetType, targetId });
    if (exist) {
      await this.favoriteEntity.delete(exist.id);
      return { favorited: false };
    }
    await this.favoriteEntity.save({ userId, targetType, targetId });
    return { favorited: true };
  }

    const qb = this.favoriteEntity
      .createQueryBuilder('f')
      .where('f.userId = :userId', { userId });
    if (query.targetType) qb.andWhere('f.targetType = :targetType', { targetType: query.targetType });
    qb.orderBy('f.id', 'DESC');
    return this.entityRenderPage(qb, query);
  }
}
```
